# Cidade 3D (`/cidade`) — Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Rota `/cidade` com um diorama 3D orbital de um centro urbano void tech onde prédios/totens abrem o conteúdo do portfolio, com um avatar low-poly autônomo correndo pela cidade.

**Architecture:** Cena procedural em @react-three/fiber num chunk lazy. `CityPage` (shell HTML: HUD, relógio, cards) segura o estado; a cena reporta hover/clique via callbacks. Dados dos lugares (posições, enquadramentos de câmera, waypoints) vivem em `places.ts`; textos em `translations.ts` (chave `city`), reusando descrições existentes.

**Tech Stack:** React 18, @react-three/fiber v8, @react-three/drei v9 (OrbitControls, Text), three 0.170, TailwindCSS + design tokens do site. `@react-three/postprocessing@^2` entra só na Task 8 (bloom desktop).

**Spec:** `docs/superpowers/specs/2026-08-25-cidade-3d-design.md`

```tsx
// src/pages/CityPage.tsx
import { lazy, Suspense, useEffect, useMemo, useState } from 'react';
import { Link } from 'react-router-dom';
import { ArrowLeft, ExternalLink, X, Clock } from 'lucide-react';
import { useLanguage } from '@/contexts/LanguageContext';
import { translations } from '@/utils/translations';
import { PLACES, type PlaceId } from '@/components/city/places';

const CityCanvas = lazy(() => import('@/components/city/CityCanvas'));

function supportsWebGL(): boolean {
  try {
    const c = document.createElement('canvas');
    return !!(c.getContext('webgl2') || c.getContext('webgl'));
  } catch {
    return false;
  }
}

/** Relógio local do visitante — só existe na cidade. Atualiza por minuto. */
const CityClock = ({ locale }: { locale: string }) => {
  const [now, setNow] = useState(() => new Date());
  useEffect(() => {
    const id = setInterval(() => setNow(new Date()), 30_000);
    return () => clearInterval(id);
  }, []);
  return (
    <span className="inline-flex items-center gap-2 px-4 py-2 rounded-full glass border border-white/10 text-sm font-semibold text-primary tabular-nums">
      <Clock className="w-4 h-4" />
      {now.toLocaleTimeString(locale, { hour: '2-digit', minute: '2-digit' })}
    </span>
  );
};

interface CardContent {
  title: string;
  subtitle?: string;
  body: string;
  tags: string[];
  live?: string;
  comingSoon?: boolean;
}

const CityPage = () => {
  const { language, toggleLanguage } = useLanguage();
  const t = translations[language];
  const tc = t.city;
  const locale = language === 'pt' ? 'pt-BR' : 'en-US';

  const [webgl] = useState(supportsWebGL);
  const [activePlace, setActivePlace] = useState<PlaceId | null>(null);
  const [hovered, setHovered] = useState<PlaceId | null>(null);
  const [interacted, setInteracted] = useState(false);
  const [reducedMotion] = useState(() =>
    window.matchMedia('(prefers-reduced-motion: reduce)').matches);
  const [isMobile] = useState(() =>
    window.matchMedia('(max-width: 767px), (pointer: coarse)').matches);

  useEffect(() => {
    const prev = document.title;
    document.title = tc.seo;
    return () => { document.title = prev; };
  }, [tc.seo]);

  // Conteúdo do card por lugar, reusando traduções existentes.
  const card: CardContent | null = useMemo(() => {
    if (!activePlace) return null;
    const place = PLACES.find((p) => p.id === activePlace);
    if (!place) return null;
    const names = tc.placeNames as Record<string, string>;
    switch (place.kind) {
      case 'project': {
        const item = t.projects.items[activePlace as keyof typeof t.projects.items];
        return { title: item.title, body: item.description, tags: [],
          live: place.live, comingSoon: place.comingSoon };
      }
      case 'experience': {
        const idx = activePlace === 'exp-twiniti' ? 0 : 1;
        const e = t.experience.list[idx];
        return { title: e.company, subtitle: `${e.title} · ${e.period}`,
          body: e.description, tags: [] };
      }
      case 'education': {
        if (activePlace === 'edu-unip') {
          const d = t.education.degree;
          return { title: d.institution, subtitle: `${d.title} · ${d.period}`,
            body: d.description, tags: [] };
        }
        return { title: 'Rocketseat', subtitle: t.education.coursesTitle,
          body: t.education.courses.map((c) => c.name).join(' · '), tags: [] };
      }
      case 'skills': {
        const key = activePlace.replace('skills-', '') as 'frontend' | 'backend' | 'tools' | 'ai';
        const stacks: Record<typeof key, string[]> = {
          frontend: ['React', 'Next.js', 'TypeScript', 'TailwindCSS', 'Vite'],
          backend: ['Java', 'Spring Boot', 'Quarkus', 'C#', 'Node.js', 'SQL'],
          tools: ['Git', 'Docker', 'Supabase', 'GitHub Actions', 'Figma'],
          ai: ['Claude Code', 'Lovable', 'Antigravity', 'Gemini Code'],
        };
        return { title: names[activePlace], subtitle: t.techStack.title,
          body: '', tags: stacks[key] };
      }
      case 'contact':
        return { title: t.contact.title, body: t.contact.subtitle, tags: [],
          live: 'https://wa.me/5512987083178' };
    }
  }, [activePlace, t, tc]);

  const hoveredName = hovered
    ? (t.projects.items as Record<string, { title: string }>)[hovered]?.title
      ?? (tc.placeNames as Record<string, string>)[hovered]
    : null;

  if (!webgl) {
    return (
      <div className="min-h-[100dvh] flex flex-col items-center justify-center gap-5 px-6 text-center">
        <p className="text-muted-foreground">{tc.noWebgl}</p>
        <Link to="/" className="inline-flex items-center gap-2 px-6 py-3 rounded-full bg-primary text-primary-foreground text-sm font-semibold">
          <ArrowLeft className="w-4 h-4" /> {tc.backToSite}
        </Link>
      </div>
    );
  }

  return (
    <div className="relative h-[100dvh] overflow-hidden bg-background">
      <Suspense fallback={null}>
        <CityCanvas
          activePlace={activePlace}
          onSelect={(id) => { setActivePlace(id); setInteracted(true); }}
          onHover={setHovered}
          reducedMotion={reducedMotion}
          isMobile={isMobile}
        />
      </Suspense>

      {/* HUD topo: voltar | relógio + idioma */}
      <header className="absolute top-0 inset-x-0 z-10 flex items-center justify-between px-4 py-4 pointer-events-none">
        <Link to="/" className="pointer-events-auto inline-flex items-center gap-2 px-4 py-2 rounded-full glass border border-white/10 text-sm text-muted-foreground hover:text-foreground transition-colors duration-200">
          <ArrowLeft className="w-4 h-4" /> {tc.backToSite}
        </Link>
        <div className="pointer-events-auto flex items-center gap-2">
          <CityClock locale={locale} />
          <button onClick={toggleLanguage}
            className="px-3.5 py-2 text-xs font-bold text-muted-foreground hover:text-foreground rounded-full glass border border-white/10 uppercase tracking-widest transition-colors duration-200">
            {language === 'pt' ? 'BR' : 'EN'}
          </button>
        </div>
      </header>

      {/* Tooltip de hover (desktop) */}
      {hoveredName && !activePlace && (
        <div className="absolute top-20 inset-x-0 z-10 flex justify-center pointer-events-none">
          <span className="px-4 py-1.5 rounded-full glass border border-primary/30 text-sm text-primary font-medium">
            {hoveredName}
          </span>
        </div>
      )}

      {/* Dica inicial */}
      {!interacted && !activePlace && (
        <div className="absolute bottom-8 inset-x-0 z-10 flex justify-center pointer-events-none">
          <span className="px-5 py-2.5 rounded-full glass border border-white/10 text-sm text-muted-foreground animate-pulse-soft">
            {tc.hint}
          </span>
        </div>
      )}

      {/* Card de conteúdo */}
      {card && (
        <div className="absolute inset-x-0 bottom-0 z-20 flex justify-center p-4 md:p-8 pointer-events-none">
          <div className="pointer-events-auto w-full max-w-md card-outer glow-primary">
            <div className="card-inner relative">
              <button onClick={() => setActivePlace(null)} aria-label={tc.close}
                className="absolute top-4 right-4 p-2 rounded-full hover:bg-white/[0.06] text-muted-foreground hover:text-foreground transition-colors duration-200">
                <X className="w-4 h-4" />
              </button>
              <h2 className="text-xl font-bold text-foreground pr-10">{card.title}</h2>
              {card.subtitle && <p className="text-sm text-primary mt-1">{card.subtitle}</p>}
              {card.body && <p className="text-sm text-muted-foreground mt-3 leading-relaxed">{card.body}</p>}
              {card.tags.length > 0 && (
                <div className="flex flex-wrap gap-2 mt-4">
                  {card.tags.map((tag) => (
                    <span key={tag} className="px-3 py-1 rounded-full glass text-xs text-muted-foreground border border-primary/10">{tag}</span>
                  ))}
                </div>
              )}
              {card.comingSoon && (
                <span className="inline-block mt-4 px-3 py-1 rounded-full bg-primary/10 border border-primary/30 text-primary text-xs font-semibold">{tc.comingSoon}</span>
              )}
              {card.live && (
                <a href={card.live} target="_blank" rel="noopener noreferrer"
                  className="inline-flex items-center gap-2 mt-4 px-5 py-2.5 rounded-full bg-primary text-primary-foreground text-sm font-semibold hover:bg-primary/90 transition-colors duration-200">
                  <ExternalLink className="w-4 h-4" /> {tc.viewLive}
                </a>
              )}
            </div>
          </div>
        </div>
      )}
    </div>
  );
};

export default CityPage;
```

```tsx
// src/components/city/CameraRig.tsx
import { useEffect, useRef } from 'react';
import { useFrame, useThree } from '@react-three/fiber';
import { OrbitControls } from '@react-three/drei';
import type { OrbitControls as OrbitControlsImpl } from 'three-stdlib';
import * as THREE from 'three';
import { HOME_FRAME, PLACES, type PlaceId } from './places';

interface Props {
  activePlace: PlaceId | null;
  reducedMotion: boolean;
}

const IDLE_MS = 10_000;

/** Órbita limitada + voo suave até o enquadramento do lugar ativo. */
const CameraRig = ({ activePlace, reducedMotion }: Props) => {
  const controls = useRef<OrbitControlsImpl>(null);
  const { camera } = useThree();
  const targetPos = useRef(new THREE.Vector3(...HOME_FRAME.position));
  const targetLook = useRef(new THREE.Vector3(...HOME_FRAME.target));
  const flying = useRef(false);
  const lastInput = useRef(Date.now());

  useEffect(() => {
    const frame = activePlace
      ? PLACES.find((p) => p.id === activePlace)!.cameraFrame
      : HOME_FRAME;
    targetPos.current.set(...frame.position);
    targetLook.current.set(...frame.target);
    flying.current = true;
  }, [activePlace]);

  useFrame((_, delta) => {
    const c = controls.current;
    if (!c) return;
    if (flying.current) {
      const k = Math.min(1, delta * 2.2);
      camera.position.lerp(targetPos.current, k);
      c.target.lerp(targetLook.current, k);
      if (camera.position.distanceTo(targetPos.current) < 0.05) flying.current = false;
    }
    // Auto-rotação lenta quando ocioso, sem lugar aberto.
    c.autoRotate = !reducedMotion && !activePlace &&
      Date.now() - lastInput.current > IDLE_MS;
    c.update();
  });

  return (
    <OrbitControls
      ref={controls}
      enableDamping
      dampingFactor={0.08}
      autoRotateSpeed={0.5}
      minDistance={10}
```

```tsx
/** Textura de janelas acesas gerada em canvas (compartilhada). */
function makeWindowTexture(): THREE.CanvasTexture {
  const c = document.createElement('canvas');
  c.width = 64; c.height = 128;
  const ctx = c.getContext('2d')!;
  ctx.fillStyle = '#0c1114';
  ctx.fillRect(0, 0, 64, 128);
  for (let y = 4; y < 124; y += 8) {
    for (let x = 4; x < 60; x += 8) {
      if (Math.random() < 0.35) {
        ctx.fillStyle = Math.random() < 0.8 ? '#1fb2aa' : '#e8f6f5';
        ctx.globalAlpha = 0.25 + Math.random() * 0.6;
        ctx.fillRect(x, y, 4, 4);
      }
    }
  }
  ctx.globalAlpha = 1;
  const tex = new THREE.CanvasTexture(c);
  tex.colorSpace = THREE.SRGBColorSpace;
  tex.wrapS = THREE.RepeatWrapping;
  tex.wrapT = THREE.RepeatWrapping;
  return tex;
}

interface Props {
  onSelect: (id: PlaceId) => void;

/**
 * Agenda de falha de neon: mostly-on com quedas breves e irregulares.
 * Seguro para fotossensibilidade: no máx ~2 quedas/s durante um "engasgo"
 * curto, e engasgos raros (a cada 6–14s por letreiro).
 */
function neonFlickerOn(t: number, seed: number): boolean {
  const cycle = 6 + (seed % 8);              // período do engasgo (6–14s)
  const local = (t + seed * 1.7) % cycle;
  if (local > 0.9) return true;              // 90% do tempo: aceso estável
  // Janela de engasgo (0.9s): 2 quedas lentas e irregulares
  return Math.sin(local * 14 + seed) > -0.35;
}

/** Um prédio-POI com letreiro neon (que ocasionalmente falha). */
const PoiBuilding = ({ place, label, hovered, onSelect, onHover, windows, reducedMotion, seed }: {
  place: Place; label: string; hovered: boolean;
  onSelect: (id: PlaceId) => void; onHover: (id: PlaceId | null) => void;
  windows: THREE.Texture; reducedMotion: boolean; seed: number;
}) => {
  const [w, h, d] = place.size;
  const sign = useRef<THREE.Group>(null);
  const body = useRef<THREE.MeshStandardMaterial>(null);

  useFrame(({ clock }) => {
    if (reducedMotion) return;
    const t = clock.elapsedTime;
    // Neon falhando (nunca no prédio em hover, para não atrapalhar a leitura)
    if (sign.current) sign.current.visible = hovered || neonFlickerOn(t, seed);
    // Janelas "respiram" dessincronizadas (cintilar lento)
    if (body.current && !hovered) {
      body.current.emissiveIntensity = 0.05 + Math.max(0, Math.sin(t * 0.6 + seed)) * 0.05;
    }
  });

  return (
    <group position={[place.position[0], 0, place.position[2]]}>
      <mesh
        position={[0, h / 2, 0]}
        onClick={(e) => { e.stopPropagation(); onSelect(place.id); }}
        onPointerOver={(e) => { e.stopPropagation(); onHover(place.id); document.body.style.cursor = 'pointer'; }}
        onPointerOut={() => { onHover(null); document.body.style.cursor = 'default'; }}
      >
        <boxGeometry args={[w, h, d]} />
        <meshStandardMaterial
          ref={body}
          color={CITY_COLORS.building}
          roughness={0.4}
          metalness={0.6}
          map={windows}
          emissive={CITY_COLORS.primary}
          emissiveIntensity={hovered ? 0.35 : 0.06}
        />
      </mesh>
      {/* Letreiro neon no topo — fixo, legível da vista padrão */}
      <group ref={sign}>
        <Text
          position={[0, h + 0.9, 0]}

          color={place.comingSoon ? CITY_COLORS.accent : CITY_COLORS.primary}
          anchorX="center" anchorY="middle"
          outlineWidth={0.02} outlineColor="#03302d"
          rotation={[0, Math.PI / 4, 0]}
        >
          {label}
        </Text>
      </group>
    </group>
  );
};

/** Prédios de fundo instanciados no anel externo (não clicáveis). */
const BackgroundBuildings = ({ count, windows }: { count: number; windows: THREE.Texture }) => {
  const ref = useRef<THREE.InstancedMesh>(null);
  const transforms = useMemo(() => {
    const m: { pos: THREE.Vector3; scale: THREE.Vector3 }[] = [];
    let seed = 42;
    const rand = () => { seed = (seed * 16807) % 2147483647; return seed / 2147483647; };
    for (let i = 0; i < count; i++) {
      const angle = (i / count) * Math.PI * 2 + rand() * 0.2;
      const radius = 30 + rand() * 11;
      const hgt = 3 + rand() * 7;
      m.push({
        pos: new THREE.Vector3(Math.cos(angle) * radius, hgt / 2, Math.sin(angle) * radius),
        scale: new THREE.Vector3(2 + rand() * 2, hgt, 2 + rand() * 2),
      });
    }
    return m;
  }, [count]);

  useFrame(() => {
    const mesh = ref.current;
    if (!mesh || mesh.userData.done) return;
    const mat = new THREE.Matrix4();
    transforms.forEach((tr, i) => {
      mat.compose(tr.pos, new THREE.Quaternion(), tr.scale);
      mesh.setMatrixAt(i, mat);
    });
    mesh.instanceMatrix.needsUpdate = true;
    mesh.userData.done = true;
  });

  return (
    <instancedMesh ref={ref} args={[undefined, undefined, count]} raycast={() => null}>
      <boxGeometry args={[1, 1, 1]} />
      <meshStandardMaterial color={CITY_COLORS.building} roughness={0.55} metalness={0.45}
        map={windows} emissive={CITY_COLORS.primary} emissiveIntensity={0.03} />
    </instancedMesh>
  );
};
```

```tsx
import * as THREE from 'three';
import { useFrame } from '@react-three/fiber';
import { Text } from '@react-three/drei';
import { AVATAR_WAYPOINTS, CITY_COLORS, TRAIN_RADIUS } from './places';

const SKIN = '#c98e63';
const RUN_SPEED = 2.2; // unidades/s

/** Avatar low-poly do João: touca preta, óculos, roupa escura, detalhe cyan. */
const Avatar = ({ reducedMotion }: { reducedMotion: boolean }) => {
  const group = useRef<THREE.Group>(null);
  const limbs = useRef<{ la?: THREE.Mesh; ra?: THREE.Mesh; ll?: THREE.Mesh; rl?: THREE.Mesh }>({});
  const state = useRef({ wp: 0, waitUntil: 0 });
  const [waving, setWaving] = useState(false);

  useFrame(({ clock }, delta) => {
    const g = group.current;
    if (!g) return;
    const t = clock.elapsedTime;
    if (reducedMotion) {
      g.position.set(...AVATAR_WAYPOINTS[0].position);
      return;
    }
    const s = state.current;
    const target = new THREE.Vector3(...AVATAR_WAYPOINTS[s.wp].position);
    const pos = g.position;
    const dist = pos.distanceTo(target);
    const paused = t < s.waitUntil;

    if (dist < 0.15 && !paused) {
      s.waitUntil = t + AVATAR_WAYPOINTS[s.wp].pause;
      s.wp = (s.wp + 1) % AVATAR_WAYPOINTS.length;
    } else if (!paused) {
      const dir = target.clone().sub(pos).normalize();
      pos.addScaledVector(dir, RUN_SPEED * delta);
      const yaw = Math.atan2(dir.x, dir.z);
      g.rotation.y += (yaw - g.rotation.y) * Math.min(1, delta * 8);
      // Ciclo de corrida: balanço de braços/pernas + bob
      const swing = Math.sin(t * 10) * 0.7;
      if (limbs.current.la) limbs.current.la.rotation.x = swing;
      if (limbs.current.ra) limbs.current.ra.rotation.x = -swing;
      if (limbs.current.ll) limbs.current.ll.rotation.x = -swing;
      if (limbs.current.rl) limbs.current.rl.rotation.x = swing;
      g.position.y = Math.abs(Math.sin(t * 10)) * 0.06;
    } else {
      // Parado "olhando vitrine": membros neutros, leve respiração
      [limbs.current.la, limbs.current.ra, limbs.current.ll, limbs.current.rl]
        .forEach((m) => { if (m) m.rotation.x *= 0.9; });
      g.position.y = Math.sin(t * 2) * 0.01;
    }
    if (waving && limbs.current.ra) limbs.current.ra.rotation.x = Math.PI + Math.sin(t * 12) * 0.4;
  });

  return (
    <group ref={group} position={AVATAR_WAYPOINTS[0].position} scale={0.55}
      onClick={(e) => { e.stopPropagation(); setWaving(true); setTimeout(() => setWaving(false), 1600); }}

        <boxGeometry args={[0.16, 0.7, 0.16]} />
        <meshStandardMaterial color="#12181b" />
      </mesh>
      <mesh ref={(m) => { limbs.current.rl = m!; }} position={[0.12, 0.35, 0]}>
        <boxGeometry args={[0.16, 0.7, 0.16]} />
        <meshStandardMaterial color="#12181b" />

      <mesh position={[0, 0.95, 0]}>
        <boxGeometry args={[0.5, 0.6, 0.3]} />
        <meshStandardMaterial color="#171f23" />
      </mesh>
      <mesh position={[0, 0.78, 0.16]}>
        <boxGeometry args={[0.5, 0.06, 0.02]} />

      </mesh>

        <meshStandardMaterial color="#171f23" />
```

```ts
// src/components/city/cityAudio.ts
// Ambiente sonoro procedural: hum grave urbano + "vento" de ruído filtrado.
// Sem arquivos de áudio. Singleton; seguro chamar start/stop repetidamente.

let ctx: AudioContext | null = null;
let master: GainNode | null = null;
let nodes: AudioNode[] = [];

function makeNoiseBuffer(c: AudioContext): AudioBuffer {
  const seconds = 2;
  const buf = c.createBuffer(1, c.sampleRate * seconds, c.sampleRate);
  const data = buf.getChannelData(0);
  for (let i = 0; i < data.length; i++) data[i] = Math.random() * 2 - 1;
  return buf;
}

export function startCityAmbience(): void {
  if (ctx) { void ctx.resume(); return; }
  ctx = new AudioContext();
  master = ctx.createGain();
  master.gain.value = 0;
  master.connect(ctx.destination);
  // Fade-in suave (2s) até volume discreto
  master.gain.linearRampToValueAtTime(0.14, ctx.currentTime + 2);

  // 1) Hum grave da cidade: dois osciladores desafinados + lowpass
  const humGain = ctx.createGain();
  humGain.gain.value = 0.5;
  const humFilter = ctx.createBiquadFilter();
  humFilter.type = 'lowpass';
  humFilter.frequency.value = 120;
  [50, 62.3].forEach((freq) => {
    const osc = ctx!.createOscillator();
    osc.type = 'triangle';
    osc.frequency.value = freq;
    osc.connect(humFilter);
    osc.start();
    nodes.push(osc);
  });
  humFilter.connect(humGain);
  humGain.connect(master);

  // 2) "Vento" urbano: ruído em loop por bandpass com LFO lento
  const noise = ctx.createBufferSource();
  noise.buffer = makeNoiseBuffer(ctx);
  noise.loop = true;
  const band = ctx.createBiquadFilter();
  band.type = 'bandpass';
  band.frequency.value = 420;
  band.Q.value = 0.6;
  const windGain = ctx.createGain();
  windGain.gain.value = 0.12;
  const lfo = ctx.createOscillator();
  lfo.frequency.value = 0.07;              // variação bem lenta
  const lfoGain = ctx.createGain();
  lfoGain.gain.value = 180;
  lfo.connect(lfoGain);
  lfoGain.connect(band.frequency);
  lfo.start();
  noise.connect(band);
  band.connect(windGain);
  windGain.connect(master);
  noise.start();
  nodes.push(noise, lfo);

  // Pausa quando a aba fica oculta
  document.addEventListener('visibilitychange', onVisibility);
}

function onVisibility(): void {
  if (!ctx) return;
  if (document.hidden) void ctx.suspend();
  else void ctx.resume();
}

export function stopCityAmbience(): void {
  if (!ctx || !master) return;
  const c = ctx;
  master.gain.linearRampToValueAtTime(0, c.currentTime + 0.6);
  window.setTimeout(() => {
    nodes.forEach((n) => { try { (n as OscillatorNode).stop?.(); } catch { /* já parado */ } });
    void c.close();
  }, 700);
  document.removeEventListener('visibilitychange', onVisibility);
  ctx = null; master = null; nodes = [];
}
```
